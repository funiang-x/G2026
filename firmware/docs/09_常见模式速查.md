## 1. 新增一个业务纯函数（`Operation/` 层）

```c
/* Operation/Inc/op_alarm.h */
#ifndef OP_ALARM_H
#define OP_ALARM_H
/* ============================================================================
 *  op_alarm.h —— 阈值报警状态机（带迟滞与去抖）
 *
 *  本文件是 Operation 层：**不许**出现 HAL / FreeRTOS / Device 的任何东西。
 *  时间由调用方以 now_ms 参数传入（见 AGENTS.md §2 R2）。
 * ==========================================================================*/
#include <stdbool.h>
#include <stdint.h>

typedef struct {
    uint16_t trip;          /* 上阈值（进入候选） */
    uint16_t clear;         /* 下阈值（退出），必须 < trip ⇒ 迟滞带 */
    uint32_t debounce_ms;   /* 候选态需连续保持多久才确认 */
    uint16_t state;         /* 内部状态，调用方不要读 */
    uint32_t since_ms;
} op_alarm_t;

void op_alarm_init(op_alarm_t *ctx, uint16_t trip, uint16_t clear, uint32_t debounce_ms);
bool op_alarm_step(op_alarm_t *ctx, uint16_t value, uint32_t now_ms);

#endif /* OP_ALARM_H */
```

```c
/* Operation/Src/op_alarm.c */
#include "op_alarm.h"

void op_alarm_init(op_alarm_t *ctx, uint16_t trip, uint16_t clear, uint32_t debounce_ms) {
    ctx->trip = trip;
    ctx->clear = clear;
    ctx->debounce_ms = debounce_ms;
    ctx->state = 0U;
    ctx->since_ms = 0U;
}

bool op_alarm_step(op_alarm_t *ctx, uint16_t value, uint32_t now_ms) {
    /* 用无符号减法算间隔 ⇒ 天然处理 2^32 回绕 */
    if (value > ctx->trip) {
        if (ctx->state == 0U) { ctx->state = 1U; ctx->since_ms = now_ms; }
        else if (now_ms - ctx->since_ms >= ctx->debounce_ms) { ctx->state = 2U; }
    } else if (value <= ctx->clear) {
        ctx->state = 0U;
    }
    return ctx->state == 2U;
}
```

**要点**：① 时间是参数；② 时间差用**无符号减法**（自动处理回绕）；
③ 参数由 `init` 注入，不读 `project_config.h`。

## 2. 给这个纯函数写 PC 单测

```c
/* Test/test_op_alarm.c */
#include "op_alarm.h"
#include "test_util.h"

static void test_hysteresis(void) {
    op_alarm_t ctx;
    op_alarm_init(&ctx, 100U, 80U, 1000U);

    CHECK(!op_alarm_step(&ctx, 50U, 0U));       /* 低于 trip */
    CHECK(!op_alarm_step(&ctx, 101U, 0U));      /* 进入候选 */
    CHECK(!op_alarm_step(&ctx, 101U, 999U));    /* 未满去抖 */
    CHECK(op_alarm_step(&ctx, 101U, 1000U));    /* 确认 */
    CHECK(op_alarm_step(&ctx, 90U, 2000U));     /* 在迟滞带内 ⇒ 仍报警 */
    CHECK(!op_alarm_step(&ctx, 79U, 3000U));    /* 低于 clear ⇒ 退出 */
}

static void test_wrap(void) {
    op_alarm_t ctx;
    op_alarm_init(&ctx, 100U, 80U, 1000U);
    op_alarm_step(&ctx, 101U, 0xFFFFFF00UL);    /* 接近回绕点 */
    CHECK(op_alarm_step(&ctx, 101U, 0x00000200UL));  /* 跨 2^32 仍算对 */
}

int main(void) {
    test_hysteresis();
    test_wrap();
    return test_report();
}
```

**要点**：每个用例都要有**负向断言**（"不该报警时确实没报"）。
只测正向的测试，等于只验证了"顺利的时候是对的"。

## 3. 中断回调投事件（`Device/` 层）

```c
/* Device/Src/bsp_uart.c */
#include "bsp_uart.h"
#include "FreeRTOS.h"
#include "queue.h"

static QueueHandle_t s_uart_evt_q;
static volatile uint8_t s_rx_byte;

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    if (huart->Instance != USART1) { return; }   /* 只认自己的实例 */
    uint8_t b = s_rx_byte;
    BaseType_t woken = pdFALSE;
    (void)xQueueSendFromISR(s_uart_evt_q, &b, &woken);
    portYIELD_FROM_ISR(woken);
    /* 立刻重新武装接收，否则只会收到一个字节 */
    (void)HAL_UART_Receive_IT(huart, (uint8_t *)&s_rx_byte, 1U);
}
```

**要点**：① 只投递不跑业务；② 认实例（一个回调服务多个 UART 时必须）；
③ 重新武装；④ `portYIELD_FROM_ISR` 不要忘。

## 4. Task 里取出事件转交 Operation

```c
/* Task/Src/app_link.c */
#include "app_link.h"
#include "op_frame.h"

void app_link_poll(uint32_t now_ms) {
    uint8_t b;
    while (xQueueReceive(s_uart_evt_q, &b, 0) == pdTRUE) {
        op_frame_feed(&s_parser, b, now_ms);     /* 纯函数，PC 上可测 */
    }
}
```

**要点**：`xQueueReceive` 的超时用 **0**（非阻塞）—— 这是"轮询式"任务，
让任务自己控制节奏；用 `portMAX_DELAY` 会把任务钉死在队列上。

## 5. 组合根装配（`Task/Src/app.c`）

★ 下面是**本工程真实的代码形状**（对照 `Task/Src/app.c` 与 `Task/Inc/app.h`）：

```c
void app_init(void) {
    bsp_log_init();                                       /* Device：日志后端，必须最先 */
    bsp_tick_init();                                      /* Device：时基 */
    app_blink_init(APP_BLINK_ON_MS, APP_BLINK_OFF_MS);     /* Operation：参数注入 */
    (void)bsp_gpio_config_out(s_led);                      /* Device：把结论落到硬件 */
    LOG_I("app_init done");
}

void app_poll(void) {                     /* 周期入口（见 §4 与 `07` 的"显式入口"） */
    app_poll_at(bsp_tick_ms());
}
```

`Core/Src/main.c` 的 `USER CODE BEGIN 2` 里**只写一行**：

```c
  /* USER CODE BEGIN 2 */
  app_init();
  /* USER CODE END 2 */
```

⚠️ **启动是另一处，不在 `main()` 里**：任务体在 `Task/Src/app_rtos.c` 的
`app_rtos_run()`（不返回），由 `Core/Src/freertos.c` 的 `StartDefaultTask` 调。
调度器本身由 `main()` 里 CubeMX 生成的 `osKernelStart()` 起 ——
所以**不要**写什么 `app_start()`。

## 7. 加一个日志

```c
#include "log.h"
LOG_I("uart ready, baud=%u", baud);
```

| 级别 | 用途 |
|---|---|
| `LOG_E` | 出错，必须处理 |
| `LOG_W` | 异常但可继续 |
| `LOG_I` | 关键状态变化（启动、连接、模式切换） |
| `LOG_D` | 调试细节，交付前可以留着（运行时按级别关） |

⚠️ **不要直接 `printf`，也不要直接调 `SEGGER_RTT_Write`** ——
通道规划集中在 `Device/Inc/rtt_channels.h`，绕过去会让通道冲突。
⚠️ **不要用 `%f`** —— 本工程链接时省掉了浮点格式化支持，打浮点是**运行期静默**输出错值。
需要打浮点就按 IEEE-754 位模式打成整数（见 `Device/Inc/bsp_log.h`）。
